$(document).ready(function() {
				   //var loader='<img src="ajax-loader(1).gif" />';
				   
				   
				   //if submit button is clicked
				 
					$('#submit').click(function () {       
						
						//show the loader
						//$('.loading').html(loader).fadeIn();     
					  
						var name = $('input[name=adminUsername]').val();
						var password = $('input[name=adminPassword]').val();
						
						 
				 
						if (name=="")
						{
							$('.message').html('Please enter the username')
						}
						else if (password=="")
						{
							$('.message').html('Please enter the password')
						}
						else
						{
						//organize the data properly
						var form_data = 
						  'adminUsername='+name+
						  '&adminPassword='+password;
						 
						//disabled all the text fields
						//$('.text').attr('disabled','true');
						 
						 
						//start the ajax
						$.ajax({
							//this is the php file that processes the data and send mail
							url: "login.php",
							 
							//POST method is used
							type: "POST",
				 
							//pass the data        
							data: form_data,    
							 
							
							//success
							success: function (html) {             
								//If not paid and application not submitted
								$('.message1').html('Login successful').fadeIn('slow');
								if (html==1) {                              
									 $('.message').hide();
									 $('.message1').html('Login successful').fadeIn('slow');
									 $('#submit').attr('disabled','true'); 
									 setTimeout(' window.location.href = "https://admissions.nwrikd.edu.ng/selectCourseAndPay/selectCourseAndPay.php"; ',1000);
								}
								//If paid but application not submitted
								else if (html==2) 
								{                              
									 $('.message').hide();
									 $('.message1').html('<i class="fa fa-check"></i>'+' Login successful').fadeIn('slow');
									 $('#submit').attr('disabled','true'); 
									 setTimeout(' window.location.href = "../Reg2/index.php"; ',1000);
								}
								//If  paid and application submitted. Print of application slip.
								else if (html==3) 
								{                              
									 $('.message').hide();
									 $('.message1').html('Login successful').fadeIn('slow');
									 $('#submit').attr('disabled','true'); 
									 setTimeout(' window.location.href = "https://admissions.nwrikd.edu.ng/Reg/printSlip.php"; ',1000);
								}
								else if (html==0)
								{
									//$('.loading').html(loader).fadeIn();
									//$('.loading').fadeOut();
									$('.message1').hide();									
									$('.message').html('Incorrect username/password').fadeIn('slow');
								}
							}
										
						});
						 }
						//cancel the submit button default behaviours
						return false;
					});
				});
